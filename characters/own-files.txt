# What each character keeps for herself: %LOCALAPPDATA%\Edgerunners Desktop\characters\<name>. Every -Character build uses these,
# so one character never reads another's memories, chats or edited personality. Settings, keys and game notes stay shared.
# Same format as app.txt, and {character} becomes her folder's name. Nothing is moved or deleted, so a character who already has
# files in the shared folder keeps them there, and they need moving by hand. Mac and Linux builds don't have this yet.

"Brain.MemoryPath = Path.Combine(data, "me.md");" => "string ownFolder = Path.Combine(Path.Combine(data, "characters"), "{character}"); Directory.CreateDirectory(ownFolder); Brain.MemoryPath = Path.Combine(ownFolder, "me.md");"
"Brain.PersonalityPath = Path.Combine(data, Brain.PersonalityName);" => "Brain.PersonalityPath = Path.Combine(ownFolder, Brain.PersonalityName);"
"Brain.MemoriesPath = Path.Combine(data, "memories.md");" => "Brain.MemoriesPath = Path.Combine(ownFolder, "memories.md");"
"Brain.TranscriptPath = Path.Combine(data, "transcripts");" => "Brain.TranscriptPath = Path.Combine(ownFolder, "transcripts");"
"brain.Load(Path.Combine(data, "chat.json"));" => "brain.Load(Path.Combine(ownFolder, "chat.json"));"

# Capri's fold-first self-test (0.8) sets the transcripts and memories paths again, inside a throwaway folder of its own.
# The two lines above match there too, so the test gets an ownFolder that is just its throwaway folder.
"string data = Path.Combine(output, "fold-first-" + Guid.NewGuid().ToString("N"));" => "string data = Path.Combine(output, "fold-first-" + Guid.NewGuid().ToString("N")); string ownFolder = data;"
