# What lets a character's sheets be drawn at double Capri's size, so they stay sharp at her largest sizes. Same format as names.txt,
# for every character. Nothing here changes how a sheet at Capri's size is drawn, and sizes can be mixed while sheets go in one at a time.

# The mouth is cut by its own width, so it still lines up when it isn't the same size as the sheet under it.
"new RectangleF(talk * width, 0, width, lips.Height)" => "new RectangleF(talk * (lips.Width / mouth.Count), 0, lips.Width / mouth.Count, lips.Height)"
# Her dancing nod bends at rows measured in 384 px art, so a taller sheet needs them scaled to find her neck.
"drop = nod * 10, lean = nod * 0.22f;" => "drop = nod * 10 * strip.Height / 384f, lean = nod * 0.22f;"
"float head = rows[0] - rowsDown, chest = rows[1] - rowsDown, neck = rows[2] - rowsDown;" => "float head = rows[0] * high / 384f - rowsDown, chest = rows[1] * high / 384f - rowsDown, neck = rows[2] * high / 384f - rowsDown;"
# On the fake crash screen she's sized from her sheet's pixels, so a double-size sheet would draw her twice as big.
"new RectangleF(LieMiddle - wide * LieSize / 2, LieTop, wide * LieSize, strip.Height * LieSize)" => "new RectangleF(LieMiddle - wide * 250f / strip.Height * LieSize / 2, LieTop, wide * 250f / strip.Height * LieSize, 250 * LieSize)"
# The self-test's colour check looks at a spot on her hair in 384 px art.
"idle.Strip.GetPixel(95, 90)" => "idle.Strip.GetPixel(95 * idle.Strip.Height / 384, 90 * idle.Strip.Height / 384)"

# The self-test pins the exact outline of Capri's art, which no one else's art can match. ci.yml still checks it on Capri's own build.
"if (idle.Strip.Width / idle.Count != 116 || idle.Strip.Height != 384 || napIdleBounds != new Rectangle(4, 20, 110, 347)" => "if (napIdleBounds.IsEmpty"
"|| laydown.Strip.Width / laydown.Count != 372 || laydown.Strip.Height != 250 || napLieBounds != new Rectangle(4, 2, 364, 245) || napSleepBounds != new Rectangle(4, 2, 357, 246)) throw" => "|| napLieBounds.IsEmpty || napSleepBounds.IsEmpty) throw"
# These lying-down checks work out her scale from pixels, so they read a sheet's height as what it is in 250 px cells.
"Math.Abs(napDrawing.Height / napDrawing.Strip.Height - LieScale / 384)" => "Math.Abs(napDrawing.Height / 250 - LieScale / 384)"
"float napPixelScale = preferences.Size * LieScale / 384;" => "float napPixelScale = preferences.Size * LieScale / 384 * 250f / napStrip.Height;"
